/* ── IBW: Devine formula. Height in inches; returns kg. ─────────────────── */
function ibwKg(sex, heightIn) {
  const base = sex === "female" ? 45.5 : 50;
  return base + 2.3 * (heightIn - 60);
}

          </p>

function Stat({ label, value, sub, strong, divider }) {
  return (
    <div style={{ flex: 1, padding: "12px 16px", borderLeft: divider ? "1px solid #c9e0db" : "none" }}>
      <div style={{ fontSize: 11.5, color: "#3f6b66", fontWeight: 600, letterSpacing: ".03em", textTransform: "uppercase" }}>
        {label}
      </div>
      <div style={{ fontSize: strong ? 22 : 19, fontWeight: 800, color: strong ? "#0a5b5b" : "#13202b",
        marginTop: 2, fontVariantNumeric: "tabular-nums" }}>
        {value}
      </div>
      {sub && <div style={{ fontSize: 11.5, color: "#3f6b66", marginTop: 1 }}>{sub}</div>}
    </div>
  );
}





import React, { useState, useMemo } from "react";

/* ============================================================================
   VENTARA — Cross-Mode Translator (v14, iteration 1)
   Scope of this iteration:
     SOURCE: generic ST  or  generic PCV   (fixed bilevel / inpatient bipap)
     TARGET: VAPS-AE PS  on ResMed Astral  (volume-targeted, auto-EPAP)

   Design intent: this is a clinical decision-support tool, not a calculator.
   Every output value declares HOW it was derived via a badge, and ambiguous
   clinical-judgment fields are forced to yellow (review/default) rather than
   being laundered into confident-looking numbers.
   ============================================================================ */

/* ---- Badge taxonomy -------------------------------------------------------
   direct      green   value carried 1:1 from a real source field
   calculated  purple  deterministic math from source inputs
   ordinal     blue    closest step on a device's ordinal scale
   default     yellow  no source equivalent — a starting bound was supplied
   review      yellow  clinical judgment required before use
   --------------------------------------------------------------------------- */
const BADGE = {
  direct:     { label: "direct",      cls: "b-direct" },
  calculated: { label: "calculated",  cls: "b-calc" },
  ordinal:    { label: "closest step",cls: "b-ord" },
  default:    { label: "default ⚠",   cls: "b-default" },
  review:     { label: "review ⚠",    cls: "b-review" },
};

/* ---- Clinical translation primitives ------------------------------------- */

// Vt (mL) -> Target Va (L/min), using the rate the volume was produced at.
// Va = (Vt - deadspace) * rate / 1000
const vtToVa = (vt, rate, deadspace = 150) =>
  ((vt - deadspace) * rate) / 1000;

// Ti from rate assuming 1:2 I:E, then min/max window.
const tiFromRate = (rate) => 60 / rate / 3;

// Rise: ResMed devices accept direct ms. Source ordinal -> ms is a v13 concern;
// for generic sources we accept ms directly and pass through.
// Trigger/Cycle: translate through L/min and % respectively (v13 layer).

/* ---- The transition engine ------------------------------------------------
   Given source inputs, produce an ordered list of target rows. Each row is
   { param, value, unit, badge, why } where `why` feeds the calc box / notes.
   --------------------------------------------------------------------------- */
function translate(src) {
  const rows = [];
  const calc = [];
  const notes = [];

  const rate = Number(src.rr);
  const ipap = Number(src.ipap);
  const epap = Number(src.epap);
  const vt = Number(src.vt);
  const isPCV = src.mode === "PCV";

  // --- EPAP min : carries directly from source EPAP ---
  rows.push({
    param: "EPAP min", value: epap, unit: "cmH₂O", badge: "direct",
    why: "Source EPAP becomes the lower bound of the auto-EPAP range.",
  });

  // --- EPAP max : no source equivalent -> clinical bound (yellow) ---
  rows.push({
    param: "EPAP max", value: epap + 4, unit: "cmH₂O", badge: "default",
    why: "Auto-EPAP needs an upper bound the fixed source never had. " +
         "Shown as EPAP min + 4 as a STARTING point only — set per the patient's " +
         "obstructive burden and titration goals.",
  });
  notes.push(
    "Fixed EPAP became an auto-adjusting range. The device will now move EPAP " +
    "between the min and max to manage upper-airway obstruction — confirm the " +
    "max is appropriate before use."
  );

  // --- Target Va : the whole point of the mode, derived from measured Vt ---
  const va = vtToVa(vt, rate);
  rows.push({
    param: "Target Va", value: round(va, 1), unit: "L/min", badge: "calculated",
    why: `Derived from the delivered Vt you entered (${vt} mL) and the set rate ` +
         `(${rate}/min): Va = (Vt − 150) × rate ÷ 1000.`,
  });
  calc.push(
    `Target Va = (${vt} − 150) × ${rate} ÷ 1000 = ${round(va, 1)} L/min`
  );
  notes.push(
    "The source mode had no volume guarantee — it delivered whatever volume the " +
    "set pressures produced. The target now guarantees ventilation to this Va, " +
    "derived from the Vt you read off the flowsheet. Treat it as a starting " +
    "target and confirm against the patient's actual ventilation on the home device."
  );

  // --- Target patient rate : we only have set rate inpatient ---
  rows.push({
    param: "Target patient rate", value: rate + 2, unit: "/min", badge: "review",
    why: "iVAPS target rate is normally patient spontaneous RR + 2. Only the set " +
         "rate is available inpatient, so this is set RR + 2 — re-check once a " +
         "spontaneous rate is observed on the home device.",
  });

  // --- PS / IPAP ceiling : source IPAP becomes the pressure-support ceiling ---
  const psMax = ipap - epap;
  rows.push({
    param: "PS max", value: psMax, unit: "cmH₂O", badge: "direct",
    why: `Source IPAP − EPAP (${ipap} − ${epap}) is the pressure support the ` +
         `patient was receiving; it becomes the ceiling the device may reach to ` +
         `hit the volume target.`,
  });
  calc.push(`PS max = IPAP − EPAP = ${ipap} − ${epap} = ${psMax} cmH₂O`);

  // --- PS min : needs a floor the source doesn't contain ---
  rows.push({
    param: "PS min", value: 4, unit: "cmH₂O", badge: "review",
    why: "VAPS needs a lower pressure-support bound the fixed source never " +
         "specified. 4 cmH₂O shown as a conservative floor — set per comfort and " +
         "the minimum support you want guaranteed.",
  });

  // --- Backup rate : carries directly ---
  rows.push({
    param: "Backup rate", value: rate, unit: "/min", badge: "direct",
    why: "Source set/backup rate carries directly.",
  });

  // --- Ti min / max ---
  const ti = isPCV ? tiFromRate(rate) : tiFromRate(rate);
  const tiMin = round(ti - 0.2, 2);
  const tiMax = round(ti + 0.6, 2);
  rows.push({
    param: "Ti min", value: tiMin, unit: "s", badge: "calculated",
    why: isPCV
      ? `Source PCV Ti is fixed; the flow-cycled target needs a window. ` +
        `Ti = 60 ÷ ${rate} ÷ 3 = ${round(ti, 2)} s, then −0.2.`
      : `ST is already flow-cycled. Ti window seeded from rate: ` +
        `Ti = 60 ÷ ${rate} ÷ 3 = ${round(ti, 2)} s, then −0.2.`,
  });
  rows.push({
    param: "Ti max", value: tiMax, unit: "s", badge: "calculated",
    why: `Ti = ${round(ti, 2)} s, then +0.6 for the upper bound.`,
  });
  calc.push(
    `Ti = 60 ÷ ${rate} ÷ 3 = ${round(ti, 2)} s  →  Ti min ${tiMin} s, Ti max ${tiMax} s`
  );
  if (isPCV) {
    notes.push(
      "The source was time-cycled (fixed Ti); the target is flow-cycled. The " +
      "single Ti became a min/max window so the device can cycle on flow within " +
      "safe time limits."
    );
  }

  // --- Trigger : through L/min (v13 common unit) ---
  if (src.trigger !== "") {
    rows.push({
      param: "Trigger", value: src.trigger, unit: "L/min", badge: "calculated",
      why: "Translated through L/min, the common trigger unit across devices.",
    });
  }

  // --- Rise : ResMed accepts ms directly ---

  }

function closestStep(scale, targetVal) {
  let best = null;
  for (const s of scale) {
    const d = Math.abs(Number(s.v) - targetVal);
    if (best === null || d < best.d) best = { ...s, d };
  }
  return best;
}

function translate(src, deviceId) {
  const dev = DEVICES[deviceId];
  const F = dev.fields;
  const caps = dev.caps;
  const rows = [], calc = [], notes = [], flags = [];

  const rate = Number(src.rr);
  const ipap = Number(src.ipap);
  const epap = Number(src.epap);
  const vt = Number(src.vt);
  const ti = Number(src.ti);
  const support = ipap - epap;
  const label = (k) => F[k] || k;

  // ===== Volume target =====
  if (F["Tidal Volume"] === "Target Va") {
    const targetRate = rate + 2;
    const va = ((vt - 150) * targetRate) / 1000;
    rows.push({ field: label("Tidal Volume"), value: round(va, 1), unit: "L/min", badge: "calculated",
      why: `Astral targets alveolar ventilation. Va = (Vt \u2212 150) \u00D7 target rate \u00F7 1000 = (${vt} \u2212 150) \u00D7 ${targetRate} \u00F7 1000.` });
    calc.push(`Target Va = (${vt} \u2212 150) \u00D7 ${targetRate} \u00F7 1000 = ${round(va,1)} L/min`);
    notes.push("This device targets alveolar ventilation (Va), not tidal volume, so the delivered Vt you entered was converted using the target rate.");
  } else {
    rows.push({ field: label("Tidal Volume"), value: vt, unit: "mL", badge: "direct",
      why: "This device targets tidal volume directly, so the delivered Vt you entered carries across as the volume target." });
    if (caps.target_volume_on === "yes")
      flags.push(`Target volume must be switched ON for ${dev.name} to guarantee this Vt.`);
  }

  // ===== EPAP =====
  const hasAutoEpap = caps.auto_epap_on === "yes";
  rows.push({ field: label("EPAP min"), value: epap, unit: "cmH\u2082O", badge: "direct",
    why: hasAutoEpap
      ? "Source EPAP becomes the lower bound of the auto-EPAP range."
      : "Source EPAP carries directly (this device has no auto-EPAP range)." });

  if (hasAutoEpap) {
    rows.push({ field: label("EPAP max"), value: epap + 4, unit: "cmH\u2082O", badge: "default",
      why: "Auto-EPAP needs an upper bound the fixed source never had. Shown as EPAP min + 4 as a STARTING point \u2014 set per the patient's obstructive burden." });
    notes.push("Fixed EPAP became an auto-adjusting range; confirm the max before use.");
    if (dev.alerts.some(a => /auto epap|intellipap/i.test(a.metric || "")))
      flags.push(`Auto-EPAP must be switched ON for ${dev.name}, or EPAP stays fixed at the min.`);
  } else {
    rows.push({ field: label("EPAP max"), value: "\u2014", unit: "", badge: "na",
      why: `${dev.name} in this mode has no auto-EPAP, so there is no EPAP max. EPAP is a single fixed value (the EPAP min above). This is a true VAPS, not auto-titrating EPAP.` });
    notes.push(`${dev.name} does not offer auto-adjusting EPAP in this mode \u2014 EPAP stays fixed at the source value. If auto-EPAP is clinically needed, choose a device that supports it (Astral, Vivo 45LS, or VPro).`);
  }

  // ===== PS band =====
  const psMax = support + 4;
  let psMin = support - 2;
  const psUnder = psMin < 0;
  if (psUnder) psMin = 0;
  rows.push({ field: label("PS max"), value: psMax, unit: "cmH\u2082O", badge: "calculated",
    why: `(IPAP \u2212 EPAP) + 4 = (${ipap} \u2212 ${epap}) + 4.` });
  rows.push({ field: label("PS min"), value: psMin, unit: "cmH\u2082O", badge: psUnder ? "review" : "calculated",
    why: psUnder
      ? `(IPAP \u2212 EPAP) \u2212 2 would be below 0 (support only ${support}); floored at 0 \u2014 confirm.`
      : `(IPAP \u2212 EPAP) \u2212 2 = (${ipap} \u2212 ${epap}) \u2212 2. Tight band around the working support.` });
  calc.push(`Support = IPAP \u2212 EPAP = ${ipap} \u2212 ${epap} = ${support} cmH\u2082O`);
  calc.push(`PS band = ${support} \u2212 2 to ${support} + 4 \u2192 PS min ${psMin}, PS max ${psMax} cmH\u2082O`);
  notes.push("Pressure support was set tightly around the support the patient was receiving inpatient (IPAP \u2212 EPAP), \u22122 to +4, to reproduce what was working.");

  // ===== Rate =====
  if (F["Respiratory Rate"].toLowerCase().includes("target rate")) {
    rows.push({ field: label("Respiratory Rate"), value: rate + 2, unit: "/min", badge: "calculated",
      why: `Astral uses a single target rate = source rate + 2 (${rate} + 2).` });
    calc.push(`Target rate = ${rate} + 2 = ${rate + 2} /min`);
  } else {
    rows.push({ field: label("Respiratory Rate"), value: rate, unit: "/min", badge: "direct",
      why: "Source set/backup rate carries directly." });
  }

  // ===== Ti =====
  if (F["Ti min"] || F["Ti max"]) {
    const tiMin = round(ti - 0.2, 1), tiMax = round(ti + 0.4, 1);
    if (F["Ti min"])
      rows.push({ field: label("Ti min"), value: tiMin, unit: "s", badge: "calculated", why: `Entered Ti ${ti} s \u2212 0.2.` });
    if (F["Ti max"])
      rows.push({ field: label("Ti max"), value: tiMax, unit: "s", badge: "calculated",
        why: `Entered Ti ${ti} s + 0.4.${F["Ti max"] === "Time Cycle" ? " On this device the time-cycle field equals Ti max." : ""}` });
    calc.push(`Ti window: ${ti} \u2212 0.2 = ${tiMin} s, ${ti} + 0.4 = ${tiMax} s`);
    if (caps.ti_min === "on/off" || caps.ti_max === "on/off")
      flags.push(`Ti min/max must be switched ON for ${dev.name} or the window won't apply.`);
  } else {
    rows.push({ field: "Ti min / Ti max", value: "\u2014", unit: "", badge: "na",
      why: `${dev.name} (${dev.mode_label}) has no Ti min/max in this mode; it uses a backup inspiratory time instead. Source Ti informs the backup Ti.` });
    if (F["Back up inspiratory time"])
      rows.push({ field: label("Back up inspiratory time"), value: ti, unit: "s", badge: "direct",
        why: "Source Ti carries to the backup inspiratory time." });
  }

  // ===== Rise =====
  const srcMs = v60RiseToMs(src.rise);
  if (/\bms\b/i.test(F["Rise"] || "")) {
    rows.push({ field: label("Rise"), value: srcMs, unit: "ms", badge: "calculated",
      why: `V60 rise ${src.rise} (1\u20136) \u2192 ${srcMs} ms; this device sets rise directly in ms.` });
    calc.push(`Rise: V60 ${src.rise} \u2192 ${srcMs} ms (direct)`);
  } else if (dev.scales.rise) {
    const step = closestStep(dev.scales.rise, srcMs);
    rows.push({ field: label("Rise"), value: step.o, unit: `(\u2248${step.v} ms)`, badge: "ordinal",
      why: `V60 rise ${src.rise} \u2248 ${srcMs} ms; closest step on ${dev.name} is ${step.o} (${step.v} ms).` });
    calc.push(`Rise: V60 ${src.rise} \u2248 ${srcMs} ms \u2192 ${dev.name} step ${step.o} (${step.v} ms)`);
  }

  // ===== Trigger =====
  const trigVal = Number(src.trigger);
  const trigScale = dev.scales.trigger || dev.scales["inspiratory trigger"] || dev.scales["inspiratory sensitivity"];
  if (trigScale) {
    const step = closestStep(trigScale, trigVal);
    rows.push({ field: label("Trigger Sensitivity"), value: step.o, unit: `(\u2248${step.v} L/min)`, badge: "ordinal",
      why: `Source trigger ${trigVal} L/min; closest step on ${dev.name} is ${step.o} (${step.v} L/min).` });
    calc.push(`Trigger: ${trigVal} L/min \u2192 ${dev.name} step ${step.o} (${step.v} L/min)`);
  } else {
    rows.push({ field: label("Trigger Sensitivity"), value: trigVal, unit: "L/min", badge: "direct",
      why: "Trigger translates through L/min, the common unit; set directly here." });
  }

  // ===== Remaining device alerts =====
  dev.alerts.forEach(a => {
    if (/intellipap/i.test(a.metric) && !flags.some(f => /intellipap|auto-epap/i.test(f)))
      flags.push(`IntelliPAP (auto-EPAP) must be switched ON for ${dev.name}.`);
    else if (/time cycle/i.test(a.metric) && /must be turned on/i.test(a.alert) && !flags.some(f => /time cycle/i.test(f)))
      flags.push(`Time Cycle must be switched ON for ${dev.name} (it acts as Ti max).`);
    else if (/does not exist/i.test(a.alert))
      notes.push(`Note for ${dev.name}: ${a.metric} \u2014 ${a.alert}.`);
  });

  return { rows, calc, notes, flags, dev };
}

    return sc.map(e=>`<option value="${e.s}">${e.s} (${e.v} L/min)</option>`).join("");

  return sc.map(e=>`<option value="${e.s}">${e.s}</option>`).join("");
}

function srcToCanon(f,raw,srcEpap){
  if(f.t==="number"){
    const v=parseFloat(raw);
    if(f.pref==="ae_ps")  return{value:v,unit:"cmH\u2082O",pref:"ae_ps"};
    if(f.pref==="above_epap"&&srcEpap!=null) return{value:v+srcEpap,unit:"cmH\u2082O",pref:"absolute",formula:`${v} + EPAP ${srcEpap} = ${v+srcEpap} cmH\u2082O`};
    if(f.pref==="absolute") return{value:v,unit:"cmH\u2082O",pref:"absolute"};
    if(f.pref==="ms")   return{value:v,unit:"ms",pref:"ms"};
    if(f.pref==="lmin") return{value:v,unit:"L/min",pref:"lmin"};
    if(f.pref==="pct")  return{value:v,unit:"%",pref:"pct"};
    return{value:v,unit:f.unit,pref:"num"};
  }
  if(f.t==="ordinal"){
    const v=exactV(raw,f.key);if(v==null)return null;
    if(f.id==="rise")    return{value:v,unit:"ms",pref:"ms"};
    if(f.id==="trigger") return{value:v,unit:"L/min",pref:"lmin"};
    if(f.id==="cycle")   return{value:v,unit:"%",pref:"pct"};
  }
  if(f.t==="verbal"){
    if(f.id==="trigger"){const v=verbalToV(raw,"ResMed_trigger_verbal");return v!=null?{value:v,unit:"L/min",pref:"lmin"}:null;}
    if(f.id==="cycle"){const k=(f.opts||[]).includes("very low")?"ResMed_cycle_verbal":"V60_cycle_verbal";const v=verbalToV(raw,k);return v!=null?{value:v,unit:"%",pref:"pct"}:null;}
  }
  return null;
}

function canonToTgt(c,tf,tgtEpap){
  if(!c||c.value==null)return null;
  if(tf.t==="number"){
    let val=c.value,formula=null;
    if(tf.pref==="ae_ps"&&c.pref==="ae_ps")   return{display:String(val),unit:tf.unit,badge:"direct"};
    if(tf.pref==="ae_ps"&&c.pref==="absolute"&&tgtEpap!=null){val=parseFloat((c.value-tgtEpap).toFixed(1));formula=`${c.value} \u2212 EPAP ${tgtEpap} = ${val} cmH\u2082O`;}
    if(tf.pref==="above_epap"&&c.pref==="absolute"&&tgtEpap!=null){val=parseFloat((c.value-tgtEpap).toFixed(1));formula=`${c.value} \u2212 EPAP ${tgtEpap} = ${val} cmH\u2082O`;}
    if(tf.pref==="absolute"&&c.pref==="ae_ps") return{display:String(c.value),unit:tf.unit,badge:"review",formula:"AE PS/PC \u2192 non-AE: verify EPAP offset."};
    if(tf.pref==="ms"&&c.pref==="ms")          return{display:String(c.value),unit:"ms",badge:"direct"};
    if(tf.pref==="lmin"&&c.pref==="lmin")      return{display:String(c.value),unit:tf.unit,badge:"direct"};
    if(tf.pref==="pct"&&c.pref==="pct")        return{display:String(c.value),unit:tf.unit,badge:"direct"};
    const display=Number.isInteger(val)?String(val):parseFloat(val.toFixed(1)).toString();
    return{display,unit:tf.unit,badge:formula?"calculated":"direct",formula};
  }
  if(tf.t==="ordinal"){
    if(tf.id==="rise"&&c.pref==="ms"){const r=closest(c.value,tf.key);if(!r)return null;return{display:String(r.s),unit:tf.unit,badge:r.approx?"closest":"exact",sub:r.approx?`approx. ${c.value}ms \u2192 nearest step ${r.s}`:""};}
    if(tf.id==="trigger"&&c.pref==="lmin"){const r=closest(c.value,tf.key);if(!r)return null;return{display:String(r.s),unit:tf.unit,badge:r.approx?"closest":"exact",sub:r.approx?`${c.value} L/min \u2192 nearest ${r.v} L/min (step ${r.s})`:""};}
    if(tf.id==="cycle"&&c.pref==="pct"){const r=closest(c.value,tf.key);if(!r)return null;return{display:String(r.s),unit:tf.unit,badge:r.approx?"closest":"exact",sub:r.approx?`wanted ${c.value}% \u2192 nearest ${r.v}%`:""};}
  }
  if(tf.t==="verbal"){
    if(tf.id==="trigger"&&c.pref==="lmin"){const r=vToVerbal(c.value,"ResMed_trigger_verbal");if(!r)return null;return{display:r.s,unit:"",badge:r.approx?"closest":"exact",sub:r.approx?`${c.value} L/min \u2192 ${r.s}`:""};}
    if(tf.id==="cycle"&&c.pref==="pct"){const k=(tf.opts||[]).includes("very low")?"ResMed_cycle_verbal":"V60_cycle_verbal";const r=vToVerbal(c.value,k);if(!r)return null;return{display:r.s,unit:"",badge:r.approx?"closest":"exact"};}
  }
  return null;
}

  const compat=COMPAT[vk][tgtDev];
  const tgtVk=compat.vk;
  const tgtConfig=DEV_MODES[tgtDev][tgtVk];
  const srcFields=DEV_MODES[srcDev][vk].fields;
  const tgtFields=tgtConfig.fields;
  const isBridge=compat.match==="bridge";
  const srcEpapRaw=getEl("epap")||getEl("epap_min");
  const srcEpap=srcEpapRaw?parseFloat(srcEpapRaw):null;
  const tgtEpap=srcEpap;
  const canon={};
  srcFields.forEach(f=>{const raw=getEl(f.id);if(!raw)return;const c=srcToCanon(f,raw,srcEpap);if(c)canon[f.id]=c;});
  const srcHasCycle=srcFields.some(f=>f.id==="cycle");
  const srcIsTimeCycled=NO_CYCLE_MODES.includes(vk)||!srcHasCycle;
  const srcHasTiMin=srcFields.some(f=>f.id==="ti_min");
  const srcHasTiMax=srcFields.some(f=>f.id==="ti_max");
  const srcHasSingleTi=srcFields.some(f=>SINGLE_TI_IDS.has(f.id));
  const tgtNeedsSingleTi=tgtFields.some(f=>SINGLE_TI_IDS.has(f.id));
  const tgtNeedsTiMin=tgtFields.some(f=>f.id==="ti_min");
  const tgtNeedsTiMax=tgtFields.some(f=>f.id==="ti_max");
  const srcIsTR=usesTR(srcDev,vk);
  const rawRR=canon.rr?canon.rr.value:null;
  const patientRRForTi=rawRR?(srcIsTR?rawRR-2:rawRR):null;
  const tiCalcNotes=[];
  if(srcHasTiMin&&srcHasTiMax&&!srcHasSingleTi&&tgtNeedsSingleTi&&patientRRForTi){
    const ct=calcTiFromRR(patientRRForTi);
    const fm=`Ti = (60 \u00f7 ${patientRRForTi} \u00f7 3) = ${ct}s (1:2 I:E from backup rate)`;
    tgtFields.filter(f=>SINGLE_TI_IDS.has(f.id)).forEach(tf=>{if(!canon[tf.id])canon[tf.id]={value:ct,unit:"s",pref:"num",derived:true,formula:fm};});
    tiCalcNotes.push(fm);
  }
  let stiVal=null;
  if(srcHasSingleTi){const sf=srcFields.find(f=>SINGLE_TI_IDS.has(f.id));if(sf&&canon[sf.id])stiVal=canon[sf.id].value;}
  if(!stiVal&&patientRRForTi)stiVal=calcTiFromRR(patientRRForTi);
  if(stiVal){
    if(tgtNeedsTiMin&&!canon.ti_min){const v=Math.max(0.1,Math.round((stiVal-0.2)*10)/10);const fm=`Ti min = ${stiVal} \u2212 0.2 = ${v}s`;canon.ti_min={value:v,unit:"s",pref:"num",derived:true,formula:fm};tiCalcNotes.push(fm);}
    if(tgtNeedsTiMax&&!canon.ti_max){const v=Math.round((stiVal+0.6)*10)/10;const fm=`Ti max = ${stiVal} + 0.6 = ${v}s`;canon.ti_max={value:v,unit:"s",pref:"num",derived:true,formula:fm};tiCalcNotes.push(fm);}
  }
  const srcIsVa=usesVa(srcDev,vk),tgtIsVa=usesVa(tgtDev,tgtVk);
  let patientRR=rawRR?(srcIsTR?rawRR-2:rawRR):null;
  if(srcIsVa&&!tgtIsVa&&canon.target_va&&patientRR!=null){
    const tr=srcIsTR?rawRR:patientRR+2;
    const vt=Math.round((canon.target_va.value/tr*1000)+150);
    canon.vt={value:vt,unit:"mL",pref:"num",derived:true,formula:`VT = (${canon.target_va.value} \u00f7 ${tr} \u00d7 1000) + 150 = ${vt} mL`};
    if(canon.rr)canon.rr={value:patientRR,unit:"bpm",pref:"num",derived:true,formula:`RR = Target Rate ${rawRR} \u2212 2 = ${patientRR} bpm`};
  }
  if(!srcIsVa&&tgtIsVa&&canon.vt&&patientRR!=null){
    const tr=patientRR+2;
    const va=parseFloat(((canon.vt.value-150)*tr/1000).toFixed(2));
    canon.target_va={value:va,unit:"L/min",pref:"num",derived:true,formula:`Va = (${canon.vt.value} \u2212 150) \u00d7 ${tr} \u00f7 1000 = ${va} L/min`};
    if(canon.rr)canon.rr={value:tr,unit:"bpm",pref:"num",derived:true,formula:`Target Rate = RR ${patientRR} + 2 = ${tr} bpm`};
  }
  const bridgeMap={};
  if(isBridge&&compat.bridges){compat.bridges.forEach(b=>{const c=canon[b.src];if(!c)return;b.tgt.forEach(tid=>{bridgeMap[tid]=Object.assign({},c,{bridgeNote:b.note||null});});});}
  const derivations=[];
  Object.values(canon).forEach(c=>{if(c&&c.formula&&!tiCalcNotes.includes(c.formula))derivations.push(c.formula);});
  tiCalcNotes.forEach(n=>{if(!derivations.includes(n))derivations.push(n);});
  const bridgeNotes=[];
  let cycleDefaulted=false;
  let rows="";
  tgtFields.forEach(tf=>{
    const isBridgedField=isBridge&&bridgeMap[tf.id]!==undefined;
    let c=isBridgedField?bridgeMap[tf.id]:canon[tf.id];
    if(tf.id==="cycle"&&!c&&srcIsTimeCycled){c={value:CYCLE_DEFAULT_PCT,unit:"%",pref:"pct",isDefault:true};cycleDefaulted=true;}
    if(isBridgedField&&bridgeMap[tf.id]&&bridgeMap[tf.id].bridgeNote&&!bridgeNotes.includes(bridgeMap[tf.id].bridgeNote))bridgeNotes.push(bridgeMap[tf.id].bridgeNote);
    let valHTML="",badgeHTML="",subHTML="";
    if(!c||c.value==null){valHTML='<span style="color:var(--text-muted)">\u2014</span>';badgeHTML='<span class="badge bw">not entered</span>';}
    else{
      const r=canonToTgt(c,tf,tgtEpap);
      if(!r){valHTML='<span style="color:var(--text-muted)">\u2014</span>';badgeHTML='<span class="badge bw">no match</span>';}
      else{
        valHTML=`<strong>${r.display}</strong>${r.unit?` <span style="color:var(--text-muted);font-size:12px">${r.unit}</span>`:""}`;
        const isDefault=c.isDefault,derived=c.derived||r.badge==="calculated",bridgePurple=isBridgedField&&!derived&&!isDefault,isReview=r.badge==="review";
        let bc,bl;
        if(isDefault){bc="bw";bl="default \u26a0";}
        else if(isReview){bc="bw";bl="review \u26a0";}
        else if(derived){bc="bp";bl="calculated";}
        else if(bridgePurple){bc="bp";bl="bridged";}
        else if(r.badge==="direct"||r.badge==="exact"){bc="bd";bl=r.badge;}
        else if(r.badge==="closest"){bc="bc";bl="closest";}
        else{bc="bw";bl=r.badge;}
        badgeHTML=`<span class="badge ${bc}">${bl}</span>`;
        if(r.sub)subHTML=`<div class="rval-sub">${r.sub}</div>`;
        if(r.formula&&!derivations.includes(r.formula))derivations.push(r.formula);
      }
    }
    rows+=`<div class="rrow"><div class="rkey">${tf.label}${tf.unit?` <span style="color:var(--text-muted);font-size:11px">(${tf.unit})</span>`:""}${tf.note?`<div class="rkey-sub">${tf.note}</div>`:""}</div><div class="rval">${valHTML}${badgeHTML}${subHTML}</div></div>`;
  });
  const uniq=[...new Set(derivations)];
  const calcBox=uniq.length?`<div class="calcbox"><strong>Calculations performed:</strong><br>${uniq.map(d=>"\u2022 "+d).join("<br>")}</div>`:"";
  const bridgeBox=(isBridge&&(bridgeNotes.length||compat.note))?`<div class="bridgebox"><strong>Parameter bridging:</strong><br>${bridgeNotes.map(n=>"\u2022 "+n).join("<br>")}${compat.note?"<br><br>"+compat.note:""}</div>`:"";
  const warnBox=cycleDefaulted?`<div class="warnbox">\u26a0 <strong>Cycle sensitivity defaulted:</strong> ${CYCLE_DEFAULT_MSG}</div>`:"";
  const pillClass=isBridge?"mpill-bridge":"mpill";
  out.innerHTML=`<div class="card">
    <div style="margin-bottom:12px"><div class="slabel" style="margin-bottom:4px">Mode on ${tgtDev}</div><span class="${pillClass}">${tgtConfig.mode}</span>${isBridge?' <span style="font-size:12px;color:var(--text-pro);margin-left:8px">parameter bridging</span>':""}</div>
    <div style="border-top:0.5px solid var(--border);padding-top:10px"><div class="slabel" style="margin-bottom:6px">Translated settings</div>${rows}</div>
    ${calcBox}${bridgeBox}${warnBox}
    <p class="disc">Purple = calculated or bridged. Blue = closest step. \u26a0 = default or review needed. Always verify clinically.</p>
  </div>`;
}
